import streamlit as st
import sqlite3
import pandas as pd
from datetime import datetime

# ============================================================
# CONFIGURATION
# ============================================================

DB_NAME = "she_system.db"

# ============================================================
# DATABASE CONNECTION
# ============================================================

def get_connection():
    return sqlite3.connect(
        DB_NAME,
        check_same_thread=False
    )

# ============================================================
# DATABASE SETUP
# ============================================================

def init_db():
    conn = get_connection()
    cursor = conn.cursor()

    # --------------------------------------------------------
    # USERS TABLE
    # --------------------------------------------------------
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            username TEXT PRIMARY KEY,
            password TEXT,
            role TEXT,
            section TEXT,
            status TEXT DEFAULT 'Pending'
        )
    """)

    # --------------------------------------------------------
    # FIX OLD DATABASES
    # Add "status" if the existing users table doesn't have it.
    # --------------------------------------------------------
    cursor.execute("PRAGMA table_info(users)")
    user_columns = [
        column[1]
        for column in cursor.fetchall()
    ]

    if "status" not in user_columns:
        cursor.execute("""
            ALTER TABLE users
            ADD COLUMN status TEXT DEFAULT 'Pending'
        """)

    # --------------------------------------------------------
    # SHE REPORTS / INCIDENTS TABLE
    # --------------------------------------------------------
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS she_reports (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp TEXT,
            username TEXT,
            section TEXT,
            hazard_type TEXT,
            description TEXT,
            severity TEXT,
            status TEXT DEFAULT 'Open'
        )
    """)

    # --------------------------------------------------------
    # FIX OLD SHE REPORTS TABLE
    # --------------------------------------------------------
    cursor.execute(
        "PRAGMA table_info(she_reports)"
    )
    report_columns = [
        column[1]
        for column in cursor.fetchall()
    ]

    if "status" not in report_columns:
        cursor.execute("""
            ALTER TABLE she_reports
            ADD COLUMN status TEXT DEFAULT 'Open'
        """)

    # --------------------------------------------------------
    # CREATE DEFAULT ADMIN
    # --------------------------------------------------------
    cursor.execute(
        "SELECT COUNT(*) FROM users"
    )
    user_count = cursor.fetchone()[0]

    if user_count == 0:
        cursor.execute("""
            INSERT INTO users
            (
                username,
                password,
                role,
                section,
                status
            )
            VALUES (?, ?, ?, ?, ?)
        """, (
            "admin",
            "admin123",
            "Overall Controller",
            "All Sections",
            "Approved"
        ))
    else:
        cursor.execute("""
            UPDATE users
            SET status = 'Approved'
            WHERE username = 'admin'
        """)

    conn.commit()
    conn.close()

# Run database initialization
init_db()

# ============================================================
# LOGIN / REGISTRATION GATE
# ============================================================

if not st.session_state.logged_in:
    st.title(
        "🛡️ CUT Digital SHE Management System"
    )
    st.caption(
        "Safety, Health & Environment Management Platform"
    )

    tab_login, tab_register = st.tabs([
        "🔑 Login",
        "📝 Request Account Access"
    ])

    # ========================================================
    # LOGIN TAB
    # ========================================================
    with tab_login:
        st.markdown(
            "### 🔐 Platform Login"
        )
        with st.form("login_form"):
            username_input = st.text_input(
                "Username"
            )
            password_input = st.text_input(
                "Password",
                type="password"
            )
            submit_btn = st.form_submit_button(
                "🚀 Login to Platform"
            )

            if submit_btn:
                if (
                    username_input.strip() == ""
                    or password_input == ""
                ):
                    st.warning(
                        "Please enter both username "
                        "and password."
                    )
                else:
                    conn = get_connection()
                    cursor = conn.cursor()
                    cursor.execute("""
                        SELECT
                            role,
                            section,
                            status
                        FROM users
                        WHERE username = ?
                        AND password = ?
                    """, (
                        username_input.strip(),
                        password_input
                    ))
                    user = cursor.fetchone()
                    conn.close()

                    if user:
                        role, section, approval_status = user
                        if approval_status == "Approved":
                            st.session_state.logged_in = True
                            st.session_state.username = (
                                username_input.strip()
                            )
                            st.session_state.role = role
                            st.session_state.section = section
                            st.success(
                                "✅ Access granted! "
                                "Loading platform..."
                            )
                            st.rerun()
                        else:
                            st.warning(
                                "⏳ Your account request is "
                                "pending approval by the "
                                "Overall Controller."
                            )
                    else:
                        st.error(
                            "❌ Invalid username or password."
                        )

    # ========================================================
    # REGISTRATION TAB
    # ========================================================
    with tab_register:
        st.markdown(
            "### 📝 Request a Section Contributor Account"
        )
        st.write(
            "Anyone can install the application, but "
            "accounts require administrator authorization "
            "before access is granted."
        )
        with st.form("reg_form"):
            new_user = st.text_input(
                "Choose Username"
            )
            new_pass = st.text_input(
                "Choose Password",
                type="password"
            )
            new_sect = st.selectbox(
                "Select Your Operational Unit / Section",
                UNIVERSITY_SECTIONS
            )
            reg_btn = st.form_submit_button(
                "📨 Submit Access Request"
            )

            if reg_btn:
                if (
                    new_user.strip() == ""
                    or new_pass.strip() == ""
                ):
                    st.warning(
                        "Please fill out all fields."
                    )
                elif len(new_user.strip()) < 3:
                    st.warning(
                        "Username must contain at least "
                        "3 characters."
                    )
                elif len(new_pass) < 4:
                    st.warning(
                        "Password must contain at least "
                        "4 characters."
                    )
                else:
                    try:
                        conn = get_connection()
                        cursor = conn.cursor()
                        cursor.execute("""
                            INSERT INTO users
                            (
                                username,
                                password,
                                role,
                                section,
                                status
                            )
                            VALUES (?, ?, ?, ?, ?)
                        """, (
                            new_user.strip(),
                            new_pass,
                            "Section Contributor",
                            new_sect,
                            "Pending"
                        ))
                        conn.commit()
                        conn.close()
                        st.success(
                            "✅ Request successfully submitted! "
                            "Please wait for the Overall Controller "
                            "to approve your account."
                        )
                    except sqlite3.IntegrityError:
                        st.error(
                            "❌ That username is already taken. "
                            "Please choose another."
                        )
                    except Exception as e:
                        st.error(
                            f"Database error: {e}"
                        )

    st.stop()
